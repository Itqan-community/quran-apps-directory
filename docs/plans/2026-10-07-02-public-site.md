```ts
import { getRequestConfig } from "next-intl/server";

export const LOCALES = ["ar", "en"] as const;
export type Locale = (typeof LOCALES)[number];
export const DEFAULT_LOCALE: Locale = "ar";

export default getRequestConfig(async ({ requestLocale }) => {
  const requested = await requestLocale;
  const locale: Locale =
    requested && (LOCALES as readonly string[]).includes(requested)
      ? (requested as Locale)
      : DEFAULT_LOCALE;

  const messages = (await import(`./${locale}.json`)).default;
  return { locale, messages };
});
```

```ts
import { notFound } from "next/navigation";
import { setRequestLocale } from "next-intl/server";
import { LOCALES, DEFAULT_LOCALE, type Locale } from "@/i18n/request";

export function isLocale(value: string): value is Locale {
  return (LOCALES as readonly string[]).includes(value);
}

/** Validate `[lang]` param, 404 on invalid, enable static rendering for next-intl. */
export function getLang(params: { lang: string }): Locale {
  if (!isLocale(params.lang)) notFound();
  setRequestLocale(params.lang);
  return params.lang;
}

export { LOCALES, DEFAULT_LOCALE };
export type { Locale };
```

```tsx
import { useEffect } from "react";

export function NavbarScrollConnector() {
  useEffect(() => {
    const header = document.getElementById("site-header");
    if (!header) return;
    const onScroll = () => {
      header.setAttribute("data-header-compact", window.scrollY > 200 ? "true" : "false");
```

- [ ] **Step 5: Write `src/components/site/compact-search.tsx`** (client) — second header row: small logo (searchLogo.svg), input (placeholder `appList.search.traditionalPlaceholder`), debounced (300ms) `searchSuggestions` call rendering a dropdown of suggestions (icon + localized name) linking to `/{lang}/app/{slug}_{id}`; Enter → navigates to `/{lang}/apps?search={q}`. Use `useTransition` + `startTransition` around the action call.

```tsx
"use client";

import { useEffect, useRef, useState, useTransition } from "react";
import Link from "next/link";
import { useTranslations } from "next-intl";
import { searchSuggestions, type Suggestion } from "@/lib/actions/search";

export function CompactSearch({ lang }: { lang: "ar" | "en" }) {
  const t = useTranslations("appList.search");
  const [query, setQuery] = useState("");
  const [items, setItems] = useState<Suggestion[]>([]);
  const [open, setOpen] = useState(false);
  const [pending, startTransition] = useTransition();
  const timer = useRef<ReturnType<typeof setTimeout> | null>(null);

  useEffect(() => {
    if (timer.current) clearTimeout(timer.current);
    if (query.trim().length < 2) { setItems([]); setOpen(false); return; }
    timer.current = setTimeout(() => {
      startTransition(async () => {
        const results = await searchSuggestions(query);
        setItems(results);
        setOpen(results.length > 0);
      });
    }, 300);
    return () => { if (timer.current) clearTimeout(timer.current); };
  }, [query]);

  function onKeyDown(e: React.KeyboardEvent<HTMLInputElement>) {
    if (e.key === "Enter" && query.trim()) {
      window.location.assign(`/${lang}/apps?search=${encodeURIComponent(query.trim())}`);
    }
    if (e.key === "Escape") setOpen(false);
  }

  return (
    <div className="compact-search-row border-t border-border bg-card/95">
      <div className="mx-auto flex max-w-7xl items-center gap-2 px-4 py-2">
        {/* eslint-disable-next-line @next/next/no-img-element */}
        <img src="/images/searchLogo.svg" alt="" width={24} height={24} />
        <div className="relative flex-1">
          <input
            value={query}
            onChange={(e) => setQuery(e.target.value)}
            onKeyDown={onKeyDown}
            onFocus={() => items.length > 0 && setOpen(true)}
            onBlur={() => setOpen(false)}
            placeholder={t("traditionalPlaceholder")}
            aria-label={t("traditional")}
            className="w-full rounded-app border border-border bg-surface px-3 py-1.5 text-sm outline-none focus:border-primary"
          />
          {open && (
            <div className="absolute inset-x-0 top-full z-50 mt-1 overflow-hidden rounded-app border border-border bg-card shadow-app">
              {pending && items.length === 0 ? (
                <div className="px-3 py-2 text-sm text-text-secondary">{t("button")}…</div>
              ) : (
                items.map((s) => (
                  <Link key={s.id} href={`/${lang}/app/${s.slug}_${s.id}`}
                    className="flex items-center gap-2 px-3 py-2 text-sm hover:bg-surface">
                    {/* eslint-disable-next-line @next/next/no-img-element */}
                    <img src={s.icon} alt="" width={24} height={24} className="rounded-app" />
                    <span>{lang === "ar" ? s.nameAr : s.nameEn}</span>
                  </Link>
                ))
              )}
            </div>
          )}
        </div>
      </div>
    </div>
  );
}
```

```tsx
"use client";

import { useRouter } from "next/navigation";
import { useTranslations } from "next-intl";
import type { Locale } from "@/lib/i18n";
import type { MetadataValuesResponse } from "@/lib/repo/metadata";
import type { CategoryItem } from "@/lib/repo/categories";
import type { AppsFilters } from "@/lib/repo/apps-rows";

export function FiltersBar({
  lang,
  current,
  categories,
  metadata,
}: {
  lang: Locale;
  current: AppsFilters;
  categories: CategoryItem[];
  metadata: MetadataValuesResponse;
}) {
  const router = useRouter();
  const t = useTranslations("appList");

  function push(next: Partial<Record<string, string[] | undefined>>) {
    const params = new URLSearchParams();
    const merged = { ...current, ...next, page: 1 };
    for (const [key, value] of Object.entries(merged)) {
      if (!value || (Array.isArray(value) && value.length === 0)) continue;
      const serialized = Array.isArray(value) ? value.join(",") : String(value);
      if (key === "status" || key === "pageSize" || key === "ordering") continue;
      params.set(key === "categorySlugs" ? "category" : key === "mushafType" ? "mushaf_type" : key, serialized);
    }
    router.push(`/${lang}/apps?${params.toString()}`);
  }

  return (
    <div className="mb-6 flex flex-col gap-3">
      <div className="flex flex-wrap gap-2">
        {categories.map((c) => {
          const active = current.categorySlugs?.includes(c.slug);
          return (
            <button key={c.id}
              onClick={() => push({ categorySlugs: active ? undefined : [c.slug] })}
              className={`rounded-full border px-3 py-1 text-sm ${active ? "border-primary bg-primary text-white" : "border-border bg-card text-text-secondary hover:border-primary"}`}>
              {lang === "ar" ? c.nameAr : c.nameEn} <span className="opacity-60">{c.appCount}</span>
            </button>
          );
        })}
      </div>
      {(["platforms", "riwayah", "mushafType", "features"] as const).map((field) => {
        const options = field === "platforms" ? metadata.platforms
          : field === "mushafType" ? metadata.mushaf_type
          : metadata[field];
        if (!options?.length) return null;
```

```tsx
import Link from "next/link";
import type { Locale } from "@/lib/i18n";

export function Pagination({
  lang, page, totalPages, searchParams,
}: {
  lang: Locale; page: number; totalPages: number;
  searchParams: Record<string, unknown>;
}) {
  if (totalPages <= 1) return null;
  function href(p: number) {
    const params = new URLSearchParams();
    for (const [k, v] of Object.entries(searchParams)) {
      if (k === "page" || !v || (Array.isArray(v) && v.length === 0)) continue;
      params.set(k, Array.isArray(v) ? v.join(",") : String(v));
    }
    if (p > 1) params.set("page", String(p));
    const qs = params.toString();
    return `/${lang}/apps${qs ? `?${qs}` : ""}`;
  }
  const pages = Array.from({ length: totalPages }, (_, i) => i + 1);
  return (
    <nav className="mt-8 flex items-center justify-center gap-2" aria-label="Pagination">
      {page > 1 && <Link href={href(page - 1)} className="rounded-app border border-border px-3 py-1.5 text-sm">{lang === "ar" ? "السابق" : "Prev"}</Link>}
      {pages.map((p) => (
        <Link key={p} href={href(p)} aria-current={p === page ? "page" : undefined}
          className={`rounded-app px-3 py-1.5 text-sm ${p === page ? "bg-primary text-white" : "border border-border text-text-secondary hover:border-primary"}`}>
          {p}
        </Link>
      ))}
      {page < totalPages && <Link href={href(page + 1)} className="rounded-app border border-border px-3 py-1.5 text-sm">{lang === "ar" ? "التالي" : "Next"}</Link>}
    </nav>
  );
```

```ts
export function parseUserAgent(ua: string | null): {
  deviceType: string;
  browser: string | null;
  os: string | null;
} {
  if (!ua) return { deviceType: "unknown", browser: null, os: null };
  const parsed = new UAParser(ua).getResult();
  const deviceType = parsed.device.type ?? "desktop";
  const normalized = ["mobile", "tablet", "desktop", "bot"].includes(deviceType) ? deviceType
    : deviceType === undefined ? "desktop"
    : "unknown";
  return {
    deviceType: normalized,
    browser: parsed.browser.name ? `${parsed.browser.name}${parsed.browser.version ? ` ${parsed.browser.version.split(".")[0]}` : ""}` : null,
    os: parsed.os.name ? `${parsed.os.name}${parsed.os.version ? ` ${parsed.os.version}` : ""}` : null,
  };
}

export function hashIp(ip: string | null): string | null {
  if (!ip) return null;
  const day = new Date().toISOString().slice(0, 10); // daily-rotating salt
  const salt = `${process.env.VIEW_IP_SALT ?? "dev-salt"}-${day}`;
  return createHash("sha256").update(`${ip}${salt}`).digest("hex");
}

export function countryFromHeaders(h: Headers): string | null {
  const cf = h.get("cf-ipcountry") ?? h.get("CF-IPCountry");

  return null;
}
```

```ts
import type { MetadataRoute } from "next";
import postgres from "postgres";

const SITE_URL = process.env.NEXT_PUBLIC_SITE_URL ?? "https://quran-apps.itqan.dev";
const LANGS = ["ar", "en"] as const;

function alternates(basePath: string) {
  return {
    ar: `${SITE_URL}/ar${basePath}`,
    en: `${SITE_URL}/en${basePath}`,
    "x-default": `${SITE_URL}/en${basePath}`,
  };
}

export default async function sitemap(): Promise<MetadataRoute.Sitemap> {
  const url = process.env.DATABASE_URL;
  if (!url) return [];
  const sql = postgres(url, { max: 2 });
  try {
    const [apps, devs] = await Promise.all([
      sql`SELECT id, slug FROM apps WHERE status = 'published' ORDER BY sort_order`,
      sql`SELECT id, name_en FROM developers`,
    ]);
    const entries: MetadataRoute.Sitemap = [];
    for (const lang of LANGS) {
      entries.push({ url: `${SITE_URL}/${lang}`, changeFrequency: "weekly", priority: 1, alternates: { languages: alternates("") } });
      entries.push({ url: `${SITE_URL}/${lang}/apps`, changeFrequency: "weekly", priority: 0.9, alternates: { languages: alternates("/apps") } });
      entries.push({ url: `${SITE_URL}/${lang}/about-us`, changeFrequency: "yearly", priority: 0.6, alternates: { languages: alternates("/about-us") } });
      entries.push({ url: `${SITE_URL}/${lang}/contact-us`, changeFrequency: "yearly", priority: 0.6, alternates: { languages: alternates("/contact-us") } });
    }
    for (const app of apps) {
      for (const lang of LANGS) {
        entries.push({ url: `${SITE_URL}/${lang}/app/${app.id}`, changeFrequency: "monthly", priority: 0.8, alternates: { languages: alternates(`/app/${app.id}`) } });
      }
    }
    for (const d of devs) {
      const slug = d.name_en.toLowerCase().replace(/\s+/g, "-");
      for (const lang of LANGS) {
        entries.push({ url: `${SITE_URL}/${lang}/developer/${slug}`, changeFrequency: "monthly", priority: 0.7, alternates: { languages: alternates(`/developer/${slug}`) } });
      }
    }
    return entries;
  } finally {
    await sql.end();
  }
}
```
